cmake_minimum_required(VERSION 3.31)
project(MeguvizMediaIdentityJNI LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID)
  message(FATAL_ERROR "This standalone JNI profile requires the Android toolchain")
endif()
option(MGM_ANDROID_COMPOSITION "Build the official single-DSO Media/RIFE/ncnn composition" OFF)
option(MGM_ANDROID_STATIC_OPENMP "Use the locked static OpenMP runtime in the official composition" ON)
option(MGM_ANDROID_SHARED_PLAN "Compile the optional shared GPU plan protocol in the official composition" ON)
if(MGM_ANDROID_COMPOSITION)
  set(MGM_ANDROID_NCNN_LAYER_PROFILE "full" CACHE STRING "ncnn layer profile for the Android composition")
  set_property(CACHE MGM_ANDROID_NCNN_LAYER_PROFILE PROPERTY STRINGS full rife46 rife46-arm64)
  option(MGM_ANDROID_PACKED_RELOCATIONS "Use Android APS2 dynamic relocations on API 23 and later" ON)
  if(NOT ANDROID_STL STREQUAL "c++_static")
    message(FATAL_ERROR "Official middleware profile owns exactly one static STL closure")
  endif()
  get_filename_component(MGM_PACK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
  set(MGM_COMPUTE_SOURCE "${MGM_PACK_ROOT}/MeguvizCompute" CACHE PATH "Compute source owner")
  set(MGM_RIFE_SOURCE "${MGM_PACK_ROOT}/MeguvizRIFE" CACHE PATH "RIFE source owner")
  set(MGC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(MGR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(MGR_BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(MGM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(MGM_WITH_RIFE ON CACHE BOOL "" FORCE)
  set(MGC_ANDROID_STATIC_OPENMP ${MGM_ANDROID_STATIC_OPENMP} CACHE BOOL "" FORCE)
  set(MGR_ENABLE_SHARED_PLAN ${MGM_ANDROID_SHARED_PLAN})
  set(model_catalog "${MGM_RIFE_SOURCE}/models/catalog/rife46.manifest.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${model_catalog}")
  file(READ "${model_catalog}" MGM_MODEL_CATALOG)
  set(MGC_NCNN_LAYER_MANIFEST "")
  set(effective_layer_profile "${MGM_ANDROID_NCNN_LAYER_PROFILE}")
  if(MGM_ANDROID_NCNN_LAYER_PROFILE STREQUAL "rife46-arm64")
    if(ANDROID_ABI STREQUAL "arm64-v8a")
      set(effective_layer_profile "rife46")
    else()
      set(effective_layer_profile "full")
    endif()
  endif()
  if(effective_layer_profile STREQUAL "rife46")
    set(MGC_NCNN_LAYER_MANIFEST "${MGM_RIFE_SOURCE}/models/catalog/rife46.ncnn-layers.json")
    file(READ "${MGC_NCNN_LAYER_MANIFEST}" layer_manifest)
    string(JSON layer_graph GET "${layer_manifest}" model_graph_sha256)
    string(JSON catalog_graph GET "${MGM_MODEL_CATALOG}" graph sha256)
    if(NOT layer_graph STREQUAL catalog_graph)
      message(FATAL_ERROR "RIFE46 ncnn layers do not match the locked model graph")
    endif()
  elseif(NOT effective_layer_profile STREQUAL "full")
    message(FATAL_ERROR "Unsupported Android ncnn layer profile: ${MGM_ANDROID_NCNN_LAYER_PROFILE}")
  endif()
  add_subdirectory("${MGM_COMPUTE_SOURCE}" compute)
  add_subdirectory("${MGM_RIFE_SOURCE}" rife)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../.." media)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/model_catalog.h" "#pragma once\ninline constexpr const char *mgm_model_catalog=R\"mgm(${MGM_MODEL_CATALOG})mgm\";\n")
  add_library(meguviz_media_rife_ncnn SHARED identity_jni.cpp frame_jni.cpp android_jni.cpp planner_jni.cpp cadence_jni.cpp)
  # The protocol remains disabled at runtime unless the host opts in.
  if(MGM_ANDROID_SHARED_PLAN)
    target_compile_definitions(meguviz_media_rife_integration PRIVATE MGM_EXPERIMENTAL_SHARED_PLAN=1)
    target_compile_definitions(meguviz_media_rife_ncnn PRIVATE MGM_EXPERIMENTAL_SHARED_PLAN=1)
  endif()
  if(MGR_ENABLE_JOB_OBSERVATION)
    target_compile_definitions(meguviz_media_rife_ncnn PRIVATE MGR_ENABLE_JOB_OBSERVATION=1)
  endif()
  target_include_directories(meguviz_media_rife_ncnn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(meguviz_media_rife_ncnn PRIVATE
    "$<LINK_LIBRARY:WHOLE_ARCHIVE,meguviz_media_android,meguviz_media_rife_integration,meguviz_media_core,meguviz_rife,meguviz_compute_core>")
  set_property(TARGET meguviz_media_rife_ncnn PROPERTY LINK_LIBRARY_OVERRIDE "WHOLE_ARCHIVE,meguviz_media_android,meguviz_media_rife_integration,meguviz_media_core,meguviz_rife,meguviz_compute_core")
  target_compile_options(meguviz_media_rife_ncnn PRIVATE -fexceptions -fvisibility=hidden -Wall -Wextra)
  set(MGM_PUBLIC_EXPORTS "{\n global:\n  meguviz_code_image_anchor_v1;\n  Java_io_github_jichuo1_meguviz_media_MediaNative_*;\n  Java_io_github_jichuo1_meguviz_media_MediaPlanner_*;\n  Java_io_github_jichuo1_meguviz_media_MediaCadence_*;\n  Java_io_github_jichuo1_meguviz_media_MediaVsyncGrid_*;\n  Java_io_github_jichuo1_meguviz_media_MediaVsyncMeter_*;\n")
  file(GLOB MGM_EXPORT_HEADERS "${MGM_COMPUTE_SOURCE}/include/meguviz/*.h" "${MGM_RIFE_SOURCE}/include/meguviz/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/../../include/meguviz/*.h")
  foreach(header IN LISTS MGM_EXPORT_HEADERS)
    file(READ "${header}" contents)
    string(REGEX MATCHALL "mg[cmr]_[A-Za-z0-9_]+[ \t\r\n]*\\(" functions "${contents}")
    foreach(function IN LISTS functions)
      string(REGEX REPLACE "[ \t\r\n]*\\(" "" function "${function}")
      if(NOT function MATCHES "^mgc_private_")
        string(APPEND MGM_PUBLIC_EXPORTS "  ${function};\n")
      endif()
    endforeach()
  endforeach()
  string(APPEND MGM_PUBLIC_EXPORTS " local: *;\n};\n")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/approved.exports.map" "${MGM_PUBLIC_EXPORTS}")
  target_link_options(meguviz_media_rife_ncnn PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/approved.exports.map" "LINKER:-z,max-page-size=16384")
  set(packed_relocations OFF)
  # The NDK legacy toolchain sets CMAKE_SYSTEM_VERSION to 1 even for API 27.
  set(android_api "${CMAKE_SYSTEM_VERSION}")
  if(DEFINED ANDROID_NATIVE_API_LEVEL)
    set(android_api "${ANDROID_NATIVE_API_LEVEL}")
  endif()
  if(MGM_ANDROID_PACKED_RELOCATIONS AND android_api GREATER_EQUAL 23)
    target_link_options(meguviz_media_rife_ncnn PRIVATE "LINKER:--pack-dyn-relocs=android")
    set(packed_relocations ON)
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/composition-options.json"
    "{\n  \"ncnn_layer_profile\": \"${MGM_ANDROID_NCNN_LAYER_PROFILE}\",\n  \"ncnn_effective_layer_profile\": \"${effective_layer_profile}\",\n  \"android_packed_relocations\": \"${packed_relocations}\"\n}\n")
  return()
endif()
set(MGM_IDENTITY_SDK_PREFIX "" CACHE PATH "Installed target-specific CPU/reference Media SDK")
set(MGM_IDENTITY_SDK_ROOT "" CACHE PATH "Parent directory with per-ABI installed SDKs")
if(MGM_IDENTITY_SDK_ROOT)
  set(MGM_IDENTITY_SDK_PREFIX "${MGM_IDENTITY_SDK_ROOT}/android-${ANDROID_ABI}/install")
endif()
if(NOT EXISTS "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizMedia/MeguvizMediaConfig.cmake")
  message(FATAL_ERROR "Run tools/build_android.py first and provide its exact ABI SDK prefix")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${MGM_IDENTITY_SDK_PREFIX}")
set(MeguvizCompute_DIR "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizCompute")
find_package(MeguvizMedia CONFIG REQUIRED PATHS "${MGM_IDENTITY_SDK_PREFIX}/lib/cmake/MeguvizMedia" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_library(meguviz_media_identity SHARED identity_jni.cpp planner_jni.cpp cadence_jni.cpp)
target_link_libraries(meguviz_media_identity PRIVATE MeguvizMedia::core)
target_compile_options(meguviz_media_identity PRIVATE -fexceptions -fvisibility=hidden)
target_link_options(meguviz_media_identity PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/identity.exports.map" "LINKER:-z,max-page-size=16384")
